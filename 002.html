const TOTAL_AYAH = 286;   // total verses

const PER_PAGE   = 20;    // verses per page

const BASE = 'https://sites.google.com/view/quransaram/Surah/002/';

const PAGE_URL    = p => `${BASE}002-${p}`;   // .../Surah/002/002-1, 002-2, ...

const SUMMARY_URL = `${BASE}002-summary`;


const grid = document.getElementById('grid');

const pages = Math.ceil(TOTAL_AYAH / PER_PAGE);

for (let p = 1; p <= pages; p++) {

  const from = (p - 1) * PER_PAGE + 1, to = Math.min(p * PER_PAGE, TOTAL_AYAH);

  const a = document.createElement('a');

  a.className = 'pg'; a.href = PAGE_URL(p); a.target = '_top';

  a.innerHTML = `<b>ഭാഗം ${p}</b><span>${from} – ${to}</span>`;

  grid.appendChild(a);

}

document.getElementById('summary').href = SUMMARY_URL;


/* ---------- Preface show / hide ---------- */

const $ = id => document.getElementById(id);

const prefToggle = $("prefToggle"), prefBody = $("prefBody");

prefToggle.addEventListener("click", () => {

  const open = prefBody.hidden;           // currently hidden -> will open

  prefBody.hidden = !open;

  prefToggle.setAttribute("aria-expanded", String(open));

  if (!open) stopSpeech();                // stop audio when hiding

});


/* ---------- Text to speech ---------- */

const SKEY = "quran-reader-settings";

const statusEl = $("status"), ttsBtn = $("ttsBtn"), prefaceEl = document.querySelector(".preface");

let token = 0, speaking = false, currentAudio = null;

const utterances = [];

const MALE_HINT = /midhun|\bmale\b|ravi|prabhat|hemant|madhur/i;

const FEMALE_HINT = /female|sobhana|swara|lekha|heera|soumya/i;


function loadSettings(){ try{ return JSON.parse(localStorage.getItem(SKEY)) || {}; }catch(e){ return {}; } }

function mlVoices(){

  const vs = window.speechSynthesis ? speechSynthesis.getVoices() : [];

  return vs.filter(v => v.lang.toLowerCase().replace("_","-").startsWith("ml"));

}

function resolveVoice(){

  const vs = mlVoices(); if (!vs.length) return null;

  const saved = loadSettings().voiceMl;   // voice chosen on chapter 1, if any

  return (saved && saved !== "auto" && vs.find(v => v.name === saved))

    || vs.find(v => MALE_HINT.test(v.name) && !FEMALE_HINT.test(v.name))

    || vs.find(v => !FEMALE_HINT.test(v.name)) || vs[0];

}

function cleanForSpeech(t){   // drop Arabic letters and brackets so they are not read wrongly

  return t.replace(/[\u0600-\u06FF\u0750-\u077F\uFB50-\uFDFF\uFE70-\uFEFF]+/g," ")

          .replace(/[()\u2018\u2019\u201C\u201D"]/g,"").replace(/\s+/g," ").trim();

}

function splitText(text, max = 150){

  const sentences = text.match(/[^.,:;!?।]+[.,:;!?।]*/g) || [text], chunks = []; let cur = "";

  for (let s of sentences){

    s = s.trim();

    while (s.length > max){

      let cut = s.lastIndexOf(" ", max); if (cut <= 0) cut = max;

      if (cur){ chunks.push(cur); cur = ""; }

      chunks.push(s.slice(0, cut).trim()); s = s.slice(cut).trim();

    }

    if (!s) continue;

    if (cur && (cur + " " + s).length > max){ chunks.push(cur); cur = s; } else cur = cur ? cur + " " + s : s;

  }

  if (cur) chunks.push(cur);

  return chunks.filter(c => /[^\s.,:;!?।]/.test(c));

}

function stopSpeech(){

  token++; speaking = false;

  if (window.speechSynthesis) speechSynthesis.cancel();

  if (currentAudio){ currentAudio.pause(); currentAudio = null; }

  ttsBtn.textContent = "▶ മുഖവുര കേൾക്കാം"; ttsBtn.classList.remove("playing");

  prefaceEl.classList.remove("reading"); statusEl.textContent = "";

}

function fail(my, msg){ if (my !== token) return; stopSpeech(); statusEl.textContent = msg; }


function playChunks(chunks, my, i = 0){   // fallback: online voice

  if (my !== token) return;

  if (i >= chunks.length){ stopSpeech(); return; }

  statusEl.textContent = "online voice " + (i + 1) + " / " + chunks.length;

  const a = new Audio("https://translate.google.com/translate_tts?ie=UTF-8&client=tw-ob&tl=ml&q=" + encodeURIComponent(chunks[i]));

  currentAudio = a; a.playbackRate = 0.92;

  a.onended = () => playChunks(chunks, my, i + 1);

  a.onerror = () => fail(my, "Voice is not available on this device or browser.");

  a.play().catch(() => fail(my, "Could not play audio. Tap the button again."));

}

function speakPreface(text, my){

  const chunks = splitText(cleanForSpeech(text)); if (!chunks.length) return;

  const voice = resolveVoice();

  if (!voice){ statusEl.textContent = "no Malayalam voice found, using online voice..."; playChunks(chunks, my); return; }

  const isMale = MALE_HINT.test(voice.name) && !FEMALE_HINT.test(voice.name);

  setTimeout(() => {   // Chrome ignores speak() right after cancel()

    if (my !== token) return;

    statusEl.textContent = voice.name; utterances.length = 0; let failed = false;

    chunks.forEach((c, idx) => {

      const u = new SpeechSynthesisUtterance(c);

      u.voice = voice; u.lang = voice.lang; u.pitch = isMale ? 1 : 0.5; u.rate = 0.95;

      u.onerror = e => {

        if (my !== token || failed || e.error === "canceled" || e.error === "interrupted") return;

        failed = true; speechSynthesis.cancel(); playChunks(chunks, my);

      };

      if (idx === chunks.length - 1) u.onend = () => { if (my === token && !failed) stopSpeech(); };

      utterances.push(u); speechSynthesis.speak(u);

    });

    speechSynthesis.resume();

  }, 100);

}

function toggleSpeech(){

  if (speaking){ stopSpeech(); return; }

  stopSpeech(); speaking = true; const my = token;

  ttsBtn.textContent = "■ നിർത്തുക"; ttsBtn.classList.add("playing"); prefaceEl.classList.add("reading");

  const text = [document.querySelector(".banner h1").textContent,

    ...[...prefaceEl.querySelectorAll(".meta, #prefBody p")].map(x => x.textContent)].join(". ");

  speakPreface(text, my);

}

ttsBtn.addEventListener("click", toggleSpeech);

if (window.speechSynthesis){ speechSynthesis.getVoices(); speechSynthesis.onvoiceschanged = () => {}; }

window.addEventListener("pagehide", stopSpeech);
