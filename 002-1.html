function cleanForSpeech(t) {

function splitText(text, max = 150) {

  const sentences = text.match(/[^.,:;!?।]+[.,:;!?।]*/g) || [text], chunks = []; let cur = "";

  for (let s of sentences) {

    s = s.trim();

    while (s.length > max) {

      let cut = s.lastIndexOf(" ", max); if (cut <= 0) cut = max;

      if (cur) { chunks.push(cur); cur = ""; }

      chunks.push(s.slice(0, cut).trim()); s = s.slice(cut).trim();

    }

    if (!s) continue;

    if (cur && (cur + " " + s).length > max) { chunks.push(cur); cur = s; } else cur = cur ? cur + " " + s : s;

  }

  if (cur) chunks.push(cur);

  return chunks.filter(c => /[^\s.,:;!?।]/.test(c));

}

    if (my !== token) return;

    if (i >= chunks.length) { done(); return; }

    statusEl.textContent = curLabel + " · " + found.voice.name + " " + (i + 1) + "/" + chunks.length;

    const u = new SpeechSynthesisUtterance(chunks[i]); utterances[0] = u;

    u.voice = found.voice; u.lang = found.voice.lang; u.pitch = found.pitch; u.rate = found.rate;

    let started = false, over = false;

    const fallback = () => { if (over) return; over = true; clearTimeout(wd); window.speechSynthesis.cancel(); playChunks(chunks, my, done, i); };

    const wd = setTimeout(() => { if (my === token && !started) fallback(); }, 5000);   // engine stuck -> online voice

  };

}

async function cachedSrc(url) {

  try {

    if (!window.caches) return url;

    if (!r) { r = await fetch(url, { mode: "cors" }); if (!r.ok) return url; await c.put(url, r.clone()); }

    return URL.createObjectURL(await r.blob());

  } catch (e) { return url; }

}
