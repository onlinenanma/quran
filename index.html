let currentQiblaBearing=0,deviceHeading=0,hasHeading=false;

function calculateQibla(lat,lon){

  const kl=21.4225*Math.PI/180,ko=39.8262*Math.PI/180,ul=lat*Math.PI/180,uo=lon*Math.PI/180,d=ko-uo;

  const y=Math.sin(d),x=Math.cos(ul)*Math.tan(kl)-Math.sin(ul)*Math.cos(d);

  return (Math.atan2(y,x)*180/Math.PI+360)%360;

}

function updateQiblaUI(lat,lon){currentQiblaBearing=calculateQibla(lat,lon);document.getElementById('qiblaAngle').innerText=`${currentQiblaBearing.toFixed(1)}°`;renderCompass();}

/* Dial rotates opposite to the phone heading so the red N pointer always shows true north;

   the gold Qibla needle sits on the dial at the Qibla bearing, so it moves with the phone. */

function renderCompass(){

  const n=document.getElementById('qiblaNeedle'),d=document.getElementById('compassDial');

  if(!d||!n)return;

  d.style.transform=`rotate(${-deviceHeading}deg)`;

  n.style.transform=`rotate(${currentQiblaBearing}deg)`;

  if(hasHeading){

    const diff=Math.abs(((currentQiblaBearing-deviceHeading+540)%360)-180);

    const ok=diff<=5;

    d.classList.toggle('aligned',ok);

    document.getElementById('compassStatus').innerText=ok?"✓ ഖിബ്‌ല ദിശയിൽ (Facing Qibla)":"Turn until the gold arrow points to the top ▼";

  }

}

function handleOrientation(e){

  let h=null;

  if(e.webkitCompassHeading!==undefined&&e.webkitCompassHeading!==null)h=e.webkitCompassHeading;

  else if(e.absolute===true||(e.alpha!==null&&e.alpha!==undefined))h=(360-e.alpha)%360;

  if(h!==null&&!isNaN(h)){

    const so=(screen.orientation&&screen.orientation.angle)||0;

    deviceHeading=(h+so)%360;hasHeading=true;renderCompass();

    const b=document.getElementById('btnStartCompass');if(b)b.style.display='none';

  }

}

async function enableLiveCompass(e){

  if(e){e.preventDefault();e.stopPropagation();}

  const st=document.getElementById('compassStatus');

  if(typeof DeviceOrientationEvent!=='undefined'&&typeof DeviceOrientationEvent.requestPermission==='function'){

    try{const r=await DeviceOrientationEvent.requestPermission();

      if(r==='granted')window.addEventListener('deviceorientation',handleOrientation,true);else st.innerText="Permission Denied";

    }catch(err){console.error(err);}

  }else if('ondeviceorientationabsolute' in window)window.addEventListener('deviceorientationabsolute',handleOrientation,true);

  else if('ondeviceorientation' in window)window.addEventListener('deviceorientation',handleOrientation,true);

  else st.innerText="Sensors Not Supported";

}

let dialAngle = 0, sinH = 0, cosH = 1, sensorOn = false;


function headingFrom(e) {                       // tilt-compensated compass heading

  if (e.webkitCompassHeading != null) return e.webkitCompassHeading;   // iPhone

  if (e.alpha == null) return null;

  if (e.beta == null || e.gamma == null) return (360 - e.alpha) % 360;

  const r = Math.PI / 180, a = e.alpha * r, b = e.beta * r, g = e.gamma * r;

  const rA = -Math.cos(a) * Math.sin(g) - Math.sin(a) * Math.sin(b) * Math.cos(g);

  const rB = -Math.sin(a) * Math.sin(g) + Math.cos(a) * Math.sin(b) * Math.cos(g);

  return (Math.atan2(rA, rB) / r + 360) % 360;

}

function setHeading(h) {                        // smooth the jitter

  const r = Math.PI / 180;

  if (!hasHeading) { sinH = Math.sin(h * r); cosH = Math.cos(h * r); }

  else { sinH = sinH * 0.75 + Math.sin(h * r) * 0.25; cosH = cosH * 0.75 + Math.cos(h * r) * 0.25; }

  deviceHeading = (Math.atan2(sinH, cosH) / r + 360) % 360;

  hasHeading = true; renderCompass();

}

function handleOrientation(e) {

  // use only true (absolute) compass data; ignore the relative event

  if (e.webkitCompassHeading == null && e.type !== 'deviceorientationabsolute' && e.absolute !== true) return;

  const h = headingFrom(e); if (h == null || isNaN(h)) return;

  const so = (screen.orientation && screen.orientation.angle) || 0;

  setHeading((h + so) % 360);

  const b = $('btnStartCompass'); if (b) b.style.display = 'none';

}

function renderCompass() {

  const n = $('qiblaNeedle'), d = $('compassDial'); if (!d || !n) return;

  dialAngle += ((((-deviceHeading - dialAngle + 180) % 360) + 360) % 360) - 180;   // shortest way round

  d.style.transform = `rotate(${dialAngle}deg)`;

  n.style.transform = `rotate(${currentQiblaBearing}deg)`;

  if (hasHeading) {

    const diff = Math.abs(((currentQiblaBearing - deviceHeading + 540) % 360) - 180), ok = diff <= 5;

    d.classList.toggle('aligned', ok);

    $('compassStatus').innerText = ok ? "✓ ഖിബ്‌ല ദിശയിൽ (Facing Qibla)" : "Turn until the gold arrow points to the top ▼";

  }

}

async function startSensors(manual) {

  const st = $('compassStatus');

  if (sensorOn) return;

  try {

    if (typeof DeviceOrientationEvent !== 'undefined' && typeof DeviceOrientationEvent.requestPermission === 'function') {

      if (!manual) return;                      // iPhone needs a tap first

      if (await DeviceOrientationEvent.requestPermission() !== 'granted') { st.innerText = "Permission Denied"; return; }

      window.addEventListener('deviceorientation', handleOrientation, true);

    } else if ('ondeviceorientationabsolute' in window) window.addEventListener('deviceorientationabsolute', handleOrientation, true);

    else if ('ondeviceorientation' in window) window.addEventListener('deviceorientation', handleOrientation, true);

    else { st.innerText = "Sensors Not Supported"; return; }

    sensorOn = true;

  } catch (err) { console.error(err); }

  setTimeout(() => { if (!hasHeading) st.innerText = "സെൻസർ ഡാറ്റ ലഭിക്കുന്നില്ല. പേജ് നേരിട്ട് തുറന്ന് നോക്കുക (Google Sites embed-ൽ തടയപ്പെടാം)."; }, 2500);

}

function enableLiveCompass(e) { if (e) { e.preventDefault(); e.stopPropagation(); } startSensors(true); }

async function initWidgets(){

  startSensors(false);

  try{

    const r=await fetch('https://ip-api.com/json/?fields=status,city,country,lat,lon'),d=await r.json();

    if(d&&d.status==='success'){matchDropdownOption(d.lat,d.lon);fetchAndDisplayPrayerTimes(d.lat,d.lon,`${d.city}, ${d.country}`);return;}

  }catch(e){}

  try{

    const r=await fetch('https://ipapi.co/json/'),d=await r.json();

    if(d&&d.latitude&&d.longitude){matchDropdownOption(d.latitude,d.longitude);fetchAndDisplayPrayerTimes(d.latitude,d.longitude,`${d.city}, ${d.country_name}`);return;}

  }catch(e){}

  document.getElementById('citySelect').value="10.8505,76.2711";

  onCityDropdownChange();

}

function requestGPSPermission(){

  if(navigator.geolocation)navigator.geolocation.getCurrentPosition(

    p=>{const la=p.coords.latitude,lo=p.coords.longitude;matchDropdownOption(la,lo);fetchAndDisplayPrayerTimes(la,lo,"GPS Location");},

    ()=>alert("Location access denied or unavailable."));

}

function matchDropdownOption(lat,lon){

  const s=document.getElementById('citySelect');

  for(const o of s.options){

    if(!o.value)continue;

    const [a,b]=o.value.split(',').map(Number);

    if(Math.abs(a-lat)<0.4&&Math.abs(b-lon)<0.4){s.value=o.value;break;}

  }

}

function onCityDropdownChange(){

  const s=document.getElementById('citySelect');

  if(!s.value)return;

  const [la,lo]=s.value.split(',');

  fetchAndDisplayPrayerTimes(parseFloat(la),parseFloat(lo),s.options[s.selectedIndex].text);

}
