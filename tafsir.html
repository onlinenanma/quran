<script>

const $ = id => document.getElementById(id);

let S = 1, A = 1, MAX = 286, trans = "", raw = "", token = 0, tfs = 20;

const utt = [];


/* HTML -> plain text, keeping paragraph breaks */

const toText = h => new DOMParser().parseFromString(String(h).replace(/<\/(p|div|h\d|li)>|<br\s*\/?>/gi, "\n"), "text/html").body.textContent || "";


/* ---------- font size (remembered) ---------- */

try { tfs = +localStorage.getItem("qr-tfs") || 20; } catch (e) {}

function fs(d) {

  tfs = Math.min(40, Math.max(16, tfs + (d || 0)));

  document.documentElement.style.setProperty("--tfs", tfs + "px");

  try { localStorage.setItem("qr-tfs", tfs); } catch (e) {}

}

fs(0);


/* ---------- verse ---------- */

async function go() {

  stop(); $("err").textContent = "";

  const s = parseInt($("sNo").value), a = parseInt($("aNo").value);

  if (!(s >= 1 && s <= 114) || !(a >= 1)) { $("err").textContent = "ദയവായി സാധുവായ സൂറത്ത്, ആയത്ത് നമ്പർ നൽകുക."; return; }

  S = s; A = a;

  try {

    const r = await fetch(`https://api.alquran.cloud/v1/ayah/${S}:${A}/editions/ar.alafasy,ml.abdulhameed`);

    const d = await r.json();

    if (d.code !== 200 || d.data.length < 2) throw 0;

    const [ar, ml] = d.data;

    MAX = ar.surah.numberOfAyahs; $("aNo").max = MAX; trans = ml.text;

    $("ref").textContent = `${ar.surah.englishName} (${ar.surah.name}) · ${S}: ${String(ar.numberInSurah).padStart(3, "0")}`;

    $("ar").textContent = ar.text; $("tr").textContent = trans; $("aud").src = ar.audio;

  } catch (e) { $("err").textContent = "വിവരങ്ങൾ ലഭ്യമാക്കാൻ സാധിച്ചില്ല. ഇന്റർനെറ്റ് പരിശോധിക്കുക."; return; }

  loadTafsir();

}

function step(d) {

  let s = S, a = A + d;

  if (a > MAX) { s++; a = 1; } else if (a < 1) { s--; a = 999; }

  if (s < 1 || s > 114) return;

  $("sNo").value = s; $("aNo").value = a; go();

  if (a === 999) setTimeout(() => { $("aNo").value = MAX; go(); }, 0);   // previous surah: jump to its last verse

}


/* ---------- tafsir ---------- */

async function loadTafsir() {

  stop(); $("tafsir").textContent = "തഫ്സീർ ലോഡ് ചെയ്യുന്നു...";

  try {

    const r = await fetch(`https://api.quran.com/api/v4/tafsirs/${$("tsel").value}/by_ayah/${S}:${A}`);

    const d = await r.json();

    raw = d && d.tafsir && d.tafsir.text ? toText(d.tafsir.text).trim() : "Tafsir content not available for this verse.";

    raw = raw.replace(/\n{3,}/g, "\n\n");

    show();

  } catch (e) { $("tafsir").textContent = "തഫ്സീർ ലോഡ് ചെയ്യുന്നതിൽ പിശക് സംഭവിച്ചു."; }

}

function show() {

  const t = $("tafsir");

  t.classList.toggle("rtl", $("tsel").value !== "169");

  t.textContent = raw;   // always the complete tafsir

}


/* ---------- text to speech ---------- */

const sc = v => (/natural|neural|enhanced|premium|online/i.test(v.name) ? 3 : 0) + (/google/i.test(v.name) ? 1 : 0) + (v.localService ? 0 : 1);

function voice(lang) {

  const vs = (window.speechSynthesis ? speechSynthesis.getVoices() : []).filter(v => v.lang.toLowerCase().replace("_", "-").startsWith(lang));

  return vs.sort((a, b) => sc(b) - sc(a))[0] || null;

}

function clean(t, lang) {

  t = toText(t);

  if (lang === "ar") t = t.replace(/[\u0610-\u061a\u064b-\u065f\u0670\u06d6-\u06ed]/g, "");   // tashkeel breaks Android TTS

  return t.replace(/[()\u2018\u2019\u201C\u201D"]/g, "").replace(/\s+/g, " ").trim();

}

function split(text, max = 150) {

  const sents = text.match(/[^.,:;!?।]+[.,:;!?।]*/g) || [text], out = []; let cur = "";

  for (let s of sents) {

    s = s.trim();

    while (s.length > max) {

      let cut = s.lastIndexOf(" ", max); if (cut <= 0) cut = max;

      if (cur) { out.push(cur); cur = ""; }

      out.push(s.slice(0, cut).trim()); s = s.slice(cut).trim();

    }

    if (!s) continue;

    if (cur && (cur + " " + s).length > max) { out.push(cur); cur = s; } else cur = cur ? cur + " " + s : s;

  }

  if (cur) out.push(cur);

  return out.filter(c => /[^\s.,:;!?।]/.test(c));

}

function stop() {

  token++; if (window.speechSynthesis) speechSynthesis.cancel();

  ["bV", "bT"].forEach(id => { $(id).textContent = "🔊 കേൾക്കാം"; $(id).classList.remove("stop"); });

  $("sV").textContent = ""; $("sT").textContent = "";

}

function say(text, lang, my, st) {

  if (!window.speechSynthesis) { st.textContent = "Speech is not supported in this browser."; return; }

  const ch = split(text), v = voice(lang);

  const next = i => {                       // one chunk at a time keeps long tafsir stable on phones

    if (my !== token) return;

    if (i >= ch.length) { stop(); return; }

    st.textContent = (v ? v.name : "default voice") + " · " + (i + 1) + "/" + ch.length;

    const u = new SpeechSynthesisUtterance(ch[i]); utt[0] = u;

    if (v) u.voice = v;

    u.lang = v ? v.lang : { ml: "ml-IN", ar: "ar-SA", en: "en-US" }[lang]; u.rate = 0.95;

    u.onend = () => next(i + 1);

    u.onerror = e => { if (my === token && e.error !== "canceled" && e.error !== "interrupted") { stop(); st.textContent = "Speech error: " + e.error; } };

    speechSynthesis.speak(u);

  };

  setTimeout(() => next(0), 100);

}

function toggle(k) {

  const b = $("b" + k);

  if (b.classList.contains("stop")) { stop(); return; }

  stop();

  const text = k === "V" ? trans : $("tafsir").textContent; if (!text) return;

  const lang = k === "V" ? "ml" : ($("tsel").value === "169" ? "en" : "ar");

  b.textContent = "⏹ നിർത്തുക"; b.classList.add("stop");

  say(clean(text, lang), lang, token, $("s" + k));

}

if (window.speechSynthesis) { speechSynthesis.getVoices(); speechSynthesis.onvoiceschanged = () => {}; }

["sNo", "aNo"].forEach(id => $(id).addEventListener("keydown", e => { if (e.key === "Enter") go(); }));

window.addEventListener("pagehide", stop);

window.onload = go;

</script>
